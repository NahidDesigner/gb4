# GB Law Firm — https://gblawfirm.com
#
# The site is open to search and to AI answer engines. For a firm whose clients
# increasingly ask an assistant before they ask a search box, being readable by
# the assistants is the point, so the major AI crawlers are named and allowed
# rather than left to the default. To reverse that for any one of them, change
# its Allow to Disallow — the blocks below exist so that is a one-line change.

User-agent: *
Allow: /

# --- Not for indexing ------------------------------------------------------
# Working files, not pages. index-v2.html and original-untouched/ are earlier
# drafts of the homepage and would compete with it as near-duplicates; the .md
# files are internal build notes; the .csv is a raw review export carrying
# reviewer names and profile links.
#
# ROBOTS.TXT IS NOT ACCESS CONTROL. It asks well-behaved crawlers not to index.
# It does not stop anyone who types the URL, and naming a path here arguably
# advertises it. These files must be excluded from the deployed directory at the
# host. Logged as a launch blocker in HANDOFF §5.
Disallow: /index-v2.html
Disallow: /original-untouched/
Disallow: /.impeccable/
Disallow: /docs/
Disallow: /styles-v2.css
Disallow: /*.md$
Disallow: /*.csv$
